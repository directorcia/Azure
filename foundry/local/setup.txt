  -Uri "http://127.0.0.1:63572/v1/models"

# foundry-prompt.ps1
# send a prompt to foundry
$body = @{
    model = "Phi-4-mini-instruct-cuda-gpu"
    messages = @(
        @{
            role = "user"
            content = "Hello"
        }
    )
} | ConvertTo-Json -Depth 10

Invoke-RestMethod `
  -Uri "http://127.0.0.1:63572/v1/chat/completions" `
  -Method Post `
  -ContentType "application/json" `
  -Body $body
