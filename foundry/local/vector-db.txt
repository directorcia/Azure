build_vectors.py

import chromadb

client = chromadb.PersistentClient(path="./vectordb")

collection = client.get_or_create_collection(
    name="knowledgebase"
)

documents = [
    "Microsoft 365 uses Entra ID for identity.",
    "The Privacy Act contains a small business exemption.",
    "Foundry Local can run models entirely on device."
]

for i, doc in enumerate(documents):
    collection.add(
        ids=[str(i)],
        documents=[doc]
    )

print("Vector database created")

    name="knowledgebase"
)

    query_texts=[

    ],
    n_results=3
